```python
class DataPoint(BaseModel):
    """Single data point in a time series."""
    date: str  # ISO date string
    value: float

class TrendMetric(BaseModel):
    """Time series for a single metric."""
    metric_name: str
    data_points: List[DataPoint]
    current_value: float
    previous_value: float
    change_percent: float
    trend_direction: str  # "up", "down", "flat"

class ActivityTrendsResponse(BaseModel):
    """Activity metrics over time."""
    calls: TrendMetric
    emails: TrendMetric
    meetings: TrendMetric
    total_activities: TrendMetric
    period: str
    granularity: str  # "day", "week", "month"
    generated_at: datetime

class ConversionTrendsResponse(BaseModel):
    """Conversion rate trends over time."""
    lead_to_qualified: TrendMetric
    qualified_to_opportunity: TrendMetric
    opportunity_to_won: TrendMetric
    period: str
    granularity: str
    generated_at: datetime

class PeriodComparisonResponse(BaseModel):
    """Period-over-period comparison."""
    current_period: Dict[str, float]
    previous_period: Dict[str, float]
    changes: Dict[str, float]  # Percentage changes
    period_type: str  # "week", "month", "quarter"
```
