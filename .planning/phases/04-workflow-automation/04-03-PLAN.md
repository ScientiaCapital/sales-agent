```python
from typing import Dict, Any
from datetime import datetime, timedelta
import logging
from app.services.crm.close_tasks import CloseTaskClient
from app.services.slack_notifier import SlackNotifier
from app.api.alerts import create_alert

logger = logging.getLogger(__name__)

class ActionExecutor:
    """Executes workflow rule actions."""

    def __init__(self, close_api_key: str, slack_webhook_url: str = None):
        self.task_client = CloseTaskClient(api_key=close_api_key)
        self.slack = SlackNotifier(webhook_url=slack_webhook_url) if slack_webhook_url else None

    async def execute(self, action: Dict[str, Any]) -> Dict[str, Any]:
        """Route action to appropriate handler."""
        action_type = action["action_type"]
        config = action["action_config"]
        context = action["context"]

        handlers = {
            "create_task": self._create_task,
            "send_alert": self._send_alert,
            "send_slack": self._send_slack,
            "trigger_agent": self._trigger_agent,
            "update_field": self._update_field,
        }

        handler = handlers.get(action_type)
        if not handler:
            raise ValueError(f"Unknown action type: {action_type}")

        result = await handler(config, context)
        await self._log_execution(action, result)
        return result

    async def _create_task(self, config: Dict, context: Dict) -> Dict:
        """Create a task in Close CRM."""
        lead_id = context.get("lead_id")
        due_days = config.get("due_days", 1)
        due_date = (datetime.now() + timedelta(days=due_days)).date()

        task = await self.task_client.create_task(
            lead_id=lead_id,
            text=config.get("task_text", "Follow up required"),
            due_date=due_date,
            task_type=config.get("task_type", "follow-up")
        )
        return {"task_id": task.get("id"), "status": "created"}

    async def _send_alert(self, config: Dict, context: Dict) -> Dict:
        """Create alert in dim_alerts table."""
        await create_alert(
            title=config.get("title", "Workflow Alert"),
            message=config.get("message", "").format(**context),
            severity=config.get("severity", "medium"),
            alert_type=config.get("alert_type", "general"),
            metadata={"rule_id": context.get("rule_id"), **context}
        )
        return {"status": "alert_created"}

    async def _send_slack(self, config: Dict, context: Dict) -> Dict:
        """Send Slack notification."""
        if not self.slack:
            return {"status": "skipped", "reason": "no_slack_configured"}
        message = config.get("message", "").format(**context)
        await self.slack.send_message(message)
        return {"status": "sent"}

    async def _trigger_agent(self, config: Dict, context: Dict) -> Dict:
        """Queue an agent task."""
        # Queue Celery task for agent execution
        from app.tasks.agent_tasks import run_agent_task
        agent_type = config.get("agent_type")
        run_agent_task.delay(agent_type, context)
        return {"status": "queued", "agent_type": agent_type}

    async def _update_field(self, config: Dict, context: Dict) -> Dict:
        """Update a field on the opportunity/lead."""
        # Use CloseProvider to update opportunity
        return {"status": "updated"}

    async def _log_execution(self, action: Dict, result: Dict) -> None:
        """Log action execution to audit trail."""
        logger.info(f"Executed action: {action['action_type']} -> {result}")
```

Update workflow tasks to execute actions after rule evaluation:

```python
from celery import shared_task
from app.services.workflow.rule_engine import WorkflowRuleEngine
from app.services.workflow.action_executor import ActionExecutor
from app.core.config import settings

@shared_task(name="evaluate_workflow_rules")
def evaluate_workflow_rules(trigger_type: str, context: dict):
    """
    Evaluate rules for a trigger and execute matching actions.
    Called by webhooks and polling tasks.
    """
    engine = WorkflowRuleEngine(db_session=get_db())
    executor = ActionExecutor(
        close_api_key=settings.CLOSE_API_KEY,
        slack_webhook_url=settings.SLACK_WEBHOOK_URL
    )

    # Evaluate rules
    actions = await engine.evaluate_event(trigger_type, context)

    # Execute all matching actions
    results = []
    for action in actions:
        try:
            result = await executor.execute(action)
            results.append({"rule_id": action["rule_id"], "result": result})
            # Update execution count on rule
            await engine.record_execution(action["rule_id"])
        except Exception as e:
            logger.error(f"Action failed: {action} - {e}")
            results.append({"rule_id": action["rule_id"], "error": str(e)})

    return {"trigger": trigger_type, "rules_matched": len(actions), "results": results}
```
